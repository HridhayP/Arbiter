cmake_minimum_required(VERSION 3.20)
project(arbiter LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()

option(ARBITER_TESTS "Build C++ unit tests (GoogleTest)" ON)
option(ARBITER_BENCH "Build C++ benchmarks (Google Benchmark)" ON)

add_library(arbiter_core STATIC cpp/src/sim.cpp)
target_include_directories(arbiter_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/cpp/include)
target_compile_options(arbiter_core PRIVATE -Wall -Wextra -Wpedantic)
set_target_properties(arbiter_core PROPERTIES POSITION_INDEPENDENT_CODE ON)

# Python extension (built by scikit-build-core during `pip install`)
if(SKBUILD)
  set(ARBITER_TESTS OFF)
  set(ARBITER_BENCH OFF)
  find_package(Python 3.10 REQUIRED COMPONENTS Interpreter Development.Module)
  find_package(pybind11 CONFIG REQUIRED)
  pybind11_add_module(_sim cpp/src/bindings.cpp)
  target_link_libraries(_sim PRIVATE arbiter_core)
  install(TARGETS _sim DESTINATION arbiter)
endif()

if(ARBITER_TESTS)
  find_package(GTest REQUIRED)
  enable_testing()
  file(GLOB TEST_SOURCES cpp/tests/*.cpp)
  add_executable(arbiter_tests ${TEST_SOURCES})
  target_link_libraries(arbiter_tests PRIVATE arbiter_core GTest::gtest GTest::gtest_main)
  include(GoogleTest)
  gtest_discover_tests(arbiter_tests)
endif()

if(ARBITER_BENCH)
  find_package(benchmark QUIET)
  if(benchmark_FOUND)
    file(GLOB BENCH_SOURCES cpp/bench/*.cpp)
    if(BENCH_SOURCES)
      add_executable(arbiter_bench ${BENCH_SOURCES})
      target_link_libraries(arbiter_bench PRIVATE arbiter_core benchmark::benchmark)
    endif()
  endif()
endif()
